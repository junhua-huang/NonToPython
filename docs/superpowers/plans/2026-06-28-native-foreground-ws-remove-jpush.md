# Native Foreground WebSocket + Remove JPush Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace JPush/vendor push with Android Native ForegroundService WebSocket delivery and remove all JPush runtime paths from frontend and backend.

**Architecture:** Flutter owns foreground real-time UI through its existing WebSocket. Android `MessageKeepAliveService` owns background delivery through OkHttp WebSocket while the foreground-service notification is visible. Backend treats Native WS as a normal `/ws?access_token=<jwt>` connection and no longer calls JPush or exposes `/api/push/*`.

**Tech Stack:** Flutter/Dart, Android Kotlin, OkHttp WebSocket, FastAPI/Python, pytest/unittest, Flutter static regression tests.

```kotlin
private fun connectWebSocket() {
    val finalUrl = normalizeWsUrl(wsUrl)
    val separator = if (finalUrl.contains("?")) "&" else "?"
    val urlWithToken = "$finalUrl${separator}access_token=$token"
    client = OkHttpClient.Builder()
        .pingInterval(25, TimeUnit.SECONDS)
        .retryOnConnectionFailure(true)
        .build()
    val request = Request.Builder().url(urlWithToken).build()
    recordDiagnostic("native_ws_last_error", "")
    webSocket = client!!.newWebSocket(request, object : WebSocketListener() {
        override fun onOpen(webSocket: WebSocket, response: Response) {
            reconnectAttempt = 0
            recordDiagnostic("native_ws_connected", true)
        }

        override fun onMessage(webSocket: WebSocket, text: String) {
            handleMessage(text)
        }

        override fun onMessage(webSocket: WebSocket, bytes: ByteString) {
            handleMessage(bytes.utf8())
        }

        override fun onClosed(webSocket: WebSocket, code: Int, reason: String) {
            recordDiagnostic("native_ws_connected", false)
            if (!stoppedByUser) scheduleReconnect("closed_$code")
        }

        override fun onFailure(webSocket: WebSocket, t: Throwable, response: Response?) {
            recordDiagnostic("native_ws_connected", false)
            recordDiagnostic("native_ws_last_error", t.message ?: t.javaClass.simpleName)
            if (!stoppedByUser) scheduleReconnect("failure")
        }
    })
}

private fun scheduleReconnect(reason: String) {
    recordDiagnostic("native_ws_last_error", reason)
    val delaySeconds = min(60, 2 + reconnectAttempt * 3)
    reconnectAttempt += 1
    mainHandler.postDelayed({
        if (!stoppedByUser && token.isNotBlank() && wsUrl.isNotBlank()) {
            connectWebSocket()
        }
    }, delaySeconds * 1000L)
}
```

```kotlin
private fun handleMessage(raw: String) {
    try {
        val root = JSONObject(raw)
        val type = root.optString("type")
        if (type == "auth_result") {
            val payload = root.optJSONObject("payload")
            val success = payload?.optBoolean("success", false) == true
            recordDiagnostic("native_ws_authenticated", success)
            if (!success) recordDiagnostic("native_ws_last_error", "auth_failed")
            return
        }
        if (type != "message") return
        val payload = root.optJSONObject("payload") ?: return
        val event = payload.optString("event")
        recordDiagnostic("native_ws_last_event", event)
        recordDiagnostic("native_ws_last_event_at", System.currentTimeMillis().toString())
        when (event) {
            "new_message" -> showMessageNotification(payload.optJSONObject("data") ?: payload)
            "new_notification" -> showInteractionNotification(payload.optJSONObject("data") ?: payload)
```
